如图为测量某种玻璃折射率的光路图。某单色光从空气垂直射入顶角为\(\alpha\)的玻璃棱镜，出射光相对于入射光的偏转角为\(\beta\)，该折射率为\paren[A]

\begin{tikzpicture}[decoration={markings,mark=at position 0.6 with {\arrow{>}}},font=\scriptsize]
	\draw (0,0)--++(80:2.5)--++(-80:2.5)--cycle;
	\draw[postaction={decorate},purple] (80:1.2)coordinate(a)--++(-10:{1.3*tan(20)})coordinate(b);
	\draw[postaction={decorate},purple] (b)--++(-40:.9)node[below right,black]{出射光};
	\draw[dashed,gray] (b)--++(-10:1);
	\draw ($(b)+(-40:.3)$) arc(-40:-10:.3);
	\draw[postaction={decorate},purple] ($(a)+(170:1)$)--node[above,black,rotate=-10]{入射光}(a);
	\draw[gray] ($(a)+(-100:.1)$)--++(-10:.1)--++(80:.1) (80:2.1) arc(260:280:.4);
    \node at ($(80:2.5)+(0,-.8)$){\(\alpha\)};
	\node at ($(b)+(-25:.7)$){\(\beta\)};
\end{tikzpicture}
\begin{choices}
	\item \(\frac{\sin(\alpha + \beta)}{\sin\alpha}\)
	\item \(\frac{\sin(\alpha + \beta)}{\sin\beta}\)
	\item \(\frac{\sin\alpha}{\sin\beta}\)
	\item \(\frac{\sin\beta}{\sin\alpha}\)
\end{choices}
