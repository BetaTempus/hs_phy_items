	\draw[->,blue] (0,0)--(0,4)node[right]{ \(y\)};
	\draw[magenta,rotate=-90] (0,0) parabola bend(0,0) (-1.5,1.5);

\begin{choices}
	\item 若船在\(x\)方向始终匀速，则船在\(y\)方向始终匀速
	\item 若船在\(x\)方向始终匀速，则船在\(y\)方向先减速后加速
	\item 若船在\(y\)方向始终匀速，则船在\(x\)方向始终匀速
	\item 若船在\(y\)方向始终匀速，则船在\(x\)方向先加速后减速
\end{choices}
