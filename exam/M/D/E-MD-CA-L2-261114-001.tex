\icode{E-MD-CA-L2-261114-001}
水平力\(F\)作用在木楔\(A\)的竖直平面上，在力\(F\)的推动下，木楔\(A\)沿斜面以恒定的加速度\(a\)向上滑动，木楔与斜面的摩擦因数为\(\mu\)，求\(F\)的大小。

\begin{tikzpicture}[>=stealth,font=\scriptsize]
	\draw[fill=teal!60] (0,0)--(37:4) (37:1.5)coordinate(a)--++(90:1)--++(-53:.8);
	\draw[->] ($(a)+(-1,.5)$)node[left]{\(F\)}--++(1,0);
	\node at($(a)+(.2,.4)$){\(A\)};
	\draw (.3,0) arc(0:37:.3);
	\node at(.5,.17){\(\theta\)};
	\foreach \x in {-.6,-.4,...,3.6}
		\draw[gray] (\x,-.2)--++(.2,.2);
	\draw[thick] (-.6,0)--(3.6,0);
\end{tikzpicture}
\begin{answer}
    \(F = \frac{m(g\sin\theta +\mu g\cos\theta + a)}{\cos\theta - \mu\sin\theta}\)
\end{answer}
