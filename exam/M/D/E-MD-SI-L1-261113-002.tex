	\item \qty{}{kg\cdot m^{-1}\cdot s^{-1}}
	\item \qty{}{kg\cdot m^{-1}\cdot s^{-2}}
	\item \qty{}{N\cdot s^{-1}}
	\item \qty{}{N\cdot m^{-1}}
\end{choices}
