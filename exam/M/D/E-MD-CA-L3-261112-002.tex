\begin{enumerate}
	\item \(A\) 与 \(B\) 上表面之间的动摩擦因数 \(\mu_1\)；
	\item \(B\) 与水平面间的动摩擦因数 \(\mu_2\)；
	\item \(A\) 的质量。
\end{enumerate}
